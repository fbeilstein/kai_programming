\subsubsection*{Mathematical part}
\textbf{Given:} An observation point in space $\mathbf{r}$, and a set of $N$ point charges with positions $\mathbf{R}_i$ and charge values $Q_i$. \\
\begin{center}

\end{center}
\textbf{Derive:} The normalized direction vector of the net electric field $\hat{E}$ at the observation point $\mathbf{r}$. \\
\textbf{Hint:} Use Coulomb's law and the superposition principle: $\vec{E}(\mathbf{r}) = \sum_{i=1}^{N} \frac{Q_i}{|\mathbf{r} - \mathbf{R}_i|^3} (\mathbf{r} - \mathbf{R}_i)$. Do not forget to normalize the resulting vector. To prevent division by zero, cap the squared distances using the predefined \lstinline|CORE_RADIUS_SQ| constant. \\

\section*{Problem 2: Field Line Trajectory (RK2 Midpoint)}

\subsubsection*{Mathematical part}
\textbf{Given:} A starting point for a streamline, the sign of the source charge $s$, and the spatial configuration of all charges. \\
\begin{center}

\end{center}
\textbf{Derive:} The iterative coordinates of the streamline utilizing the 2nd-Order Runge-Kutta (Midpoint) method. \\
\textbf{Hint:} For each step of size $h$, first scout the midpoint $\mathbf{r}_{mid} = \mathbf{r}_n + \frac{h}{2} \hat{E}(\mathbf{r}_n) s$. Then, use the field direction at the midpoint to take the full step $\mathbf{r}_{n+1} = \mathbf{r}_n + h \hat{E}(\mathbf{r}_{mid}) s$. The integration must stop early if the line enters the \lstinline|CORE_RADIUS_SQ| of any charge. \\
